```typescript
// src/app/api/auth/google/initiate/route.ts
// Initiates Google OAuth2 flow for Calendar integration.
// D-02: Generates random state token, stores in httpOnly cookie, passes as OAuth state param.
// D-04: Uses googleapis OAuth2Client directly.
// Scope: calendar.events (restricted — requires CASA Tier 2 verification for production).
// NOTE: prompt:'consent' REQUIRED — without it, re-auth returns refresh_token:null from getToken().
import { cookies } from 'next/headers'
import { NextResponse } from 'next/server'
import { randomBytes } from 'crypto'
import { createOAuth2Client } from '@/lib/google/client'

export async function GET() {
  const state = randomBytes(32).toString('hex')
  const oauth2Client = createOAuth2Client()

  const authUrl = oauth2Client.generateAuthUrl({
    access_type: 'offline',    // REQUIRED — gets refresh_token for server-side use
    prompt: 'consent',         // REQUIRED — ensures fresh refresh_token on re-auth (Risk 1)
    scope: ['https://www.googleapis.com/auth/calendar.events'],
    state,
  })

  // D-02: Set httpOnly state cookie before redirect.
  // sameSite:'lax' (NOT 'strict') — OAuth callback is a cross-site GET redirect; 'strict' blocks the cookie.
  const cookieStore = await cookies()
  cookieStore.set('oauth_state', state, {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'lax',
    maxAge: 600,  // 10 minutes — D-02 state cookie TTL
    path: '/',
  })

  return NextResponse.redirect(authUrl)
}
```

```typescript
// src/app/api/auth/google/callback/route.ts
// Handles Google OAuth2 callback: verifies state, exchanges code, encrypts + stores tokens.
// D-02: Verifies oauth_state cookie matches state param before any DB interaction.
// D-03: Redirects to /dashboard?calendarConnected=1 on success.
// SECURITY: contractor_id derived from JWT user.id — NEVER from URL params (STRIDE: EoP).
// CRITICAL: tokens.expiry_date from googleapis is milliseconds since epoch — NOT seconds.
//   Convert with new Date(tokens.expiry_date) for Supabase TIMESTAMPTZ.
import { cookies } from 'next/headers'
import { NextResponse } from 'next/server'
import { createOAuth2Client } from '@/lib/google/client'
import { createServiceRoleClient } from '@/lib/supabase/service'
import { createSupabaseServer } from '@/lib/supabase/server'
import { encryptToken } from '@/lib/crypto/tokens'

export async function GET(req: Request) {
  const url = new URL(req.url)
  const code = url.searchParams.get('code')
  const returnedState = url.searchParams.get('state')

  // D-02: Verify state cookie FIRST — before any DB interaction (CSRF prevention)
  const cookieStore = await cookies()
  const expectedState = cookieStore.get('oauth_state')?.value
  if (!expectedState || returnedState !== expectedState) {
    return NextResponse.redirect(new URL('/dashboard?calendarError=state_mismatch', req.url))
  }
  cookieStore.delete('oauth_state')

  if (!code) {
    return NextResponse.redirect(new URL('/dashboard?calendarError=no_code', req.url))
  }

  // Auth check — getUser() validates JWT; never use getSession()
  const supabase = await createSupabaseServer()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.redirect(new URL('/auth/login', req.url))

  // Derive contractor_id from JWT user.id — NEVER accept from URL params (STRIDE: Elevation of Privilege)
  const db = createServiceRoleClient()
  const { data: contractor } = await db
    .from('contractors')
    .select('id')
    .eq('user_id', user.id)
    .maybeSingle()
  if (!contractor) {
    return NextResponse.redirect(new URL('/dashboard?calendarError=no_contractor', req.url))
  }

  try {
    const oauth2Client = createOAuth2Client()
    const { tokens } = await oauth2Client.getToken(code)

    // tokens.expiry_date is milliseconds since epoch (Risk 2: NOT seconds).
    // new Date(ms) is correct; new Date(ms * 1000) would be wrong.
    if (!tokens.access_token || !tokens.refresh_token) {
      // refresh_token is null if prompt:'consent' was omitted or token was not re-granted
      return NextResponse.redirect(new URL('/dashboard?calendarError=missing_tokens', req.url))
    }

    await db.from('calendar_credentials').upsert({
      contractor_id: contractor.id,
      access_token: encryptToken(tokens.access_token),
      refresh_token: encryptToken(tokens.refresh_token),
      token_expiry: new Date(tokens.expiry_date!).toISOString(),
      scope: tokens.scope ?? null,
    }, { onConflict: 'contractor_id' })

    // D-03: Redirect to dashboard with success param
    return NextResponse.redirect(new URL('/dashboard?calendarConnected=1', req.url))
  } catch (err) {
    console.error('[auth/google/callback] token exchange failed:', err instanceof Error ? err.message : err)
    return NextResponse.redirect(new URL('/dashboard?calendarError=token_exchange_failed', req.url))
  }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → /api/auth/google/initiate | Initiates OAuth; state token prevents CSRF |
| Google → /api/auth/google/callback | OAuth code delivery; untrusted query params (state, code) |
| callback route → calendar_credentials (Supabase) | Writes encrypted tokens via service role |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-06-01 | Spoofing | CSRF on OAuth callback | mitigate | state param + httpOnly cookie (D-02): expectedState verified against returnedState before any DB interaction; cookie deleted after verification to prevent replay |
| T-04-06-02 | Elevation of Privilege | contractor_id from URL params | mitigate | callback route derives contractor_id from user.id (JWT via getUser()) — never from code, state, or any URL param |
| T-04-06-03 | Information Disclosure | plaintext OAuth tokens in DB | mitigate | encryptToken() (AES-256-GCM) applied to both access_token and refresh_token before upsert; never logged |
| T-04-06-04 | Information Disclosure | token leak via error messages | mitigate | catch block logs only err.message (no token values); error redirects use generic error codes (state_mismatch, token_exchange_failed) — no token values in URL |
| T-04-06-05 | Tampering | missing prompt:'consent' on re-auth | mitigate | prompt: 'consent' hardcoded in generateAuthUrl() — explicit code comment warns against removing; refresh_token null check in callback route catches the case and redirects with error |
| T-04-06-06 | Denial of Service | oauth_state cookie not deleted after success | mitigate | cookieStore.delete('oauth_state') called immediately after verification and before any DB interaction |
</threat_model>
