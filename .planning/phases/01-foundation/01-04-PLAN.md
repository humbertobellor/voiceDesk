    2. **Create `middleware.ts`** at the project root (NOT in src/) — this is Next.js's required location. Copy from RESEARCH.md Pattern 2 lines 268–326 with the Pitfall 4 fix (handle both "no contractor row" and "contractor row with null phone_number"):

    ```typescript
    // middleware.ts
    // Source: RESEARCH.md Pattern 2 + Pitfall 4 (phone capture bypass).
    // CRITICAL: getUser() not getSession() — see RESEARCH.md Pitfall 2 + Anti-Patterns.
    import { NextResponse } from 'next/server'
    import type { NextRequest } from 'next/server'
    import { createServerClient } from '@supabase/ssr'

    export async function middleware(request: NextRequest) {
      const response = NextResponse.next({ request: { headers: request.headers } })

      const supabase = createServerClient(
        process.env.NEXT_PUBLIC_SUPABASE_URL!,
        process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
        {
          cookies: {
            getAll: () => request.cookies.getAll(),
            // CRITICAL: write cookies to BOTH request and response (Pitfall 3)
            setAll: (cookies) => {
              cookies.forEach(({ name, value, options }) => {
                request.cookies.set(name, value)
                response.cookies.set(name, value, options)
              })
            },
          },
        }
      )

      // MUST call getUser() to refresh session (Pitfall 2)
      const { data: { user } } = await supabase.auth.getUser()
      const path = request.nextUrl.pathname

      // Unauthenticated users hit /dashboard or /setup → /auth/login
      if (!user && (path.startsWith('/dashboard') || path.startsWith('/setup'))) {
        return NextResponse.redirect(new URL('/auth/login', request.url))
      }

      // Authenticated users on /auth/* (login/signup pages) → /dashboard if they have a phone, else /setup/phone
      if (user && path.startsWith('/auth/') && !path.startsWith('/auth/callback')) {
        const { data: contractor } = await supabase
          .from('contractors')
          .select('phone_number')
          .eq('user_id', user.id)
          .maybeSingle()
        if (!contractor?.phone_number) {
          return NextResponse.redirect(new URL('/setup/phone', request.url))
        }
        return NextResponse.redirect(new URL('/dashboard', request.url))
      }

      // Phone capture gate (D-02 + Pitfall 4): authenticated user accessing /dashboard
      // must have a contractors row with phone_number set. Both "no row" and "row with null phone"
      // route to /setup/phone.
      if (user && path.startsWith('/dashboard')) {
        const { data: contractor } = await supabase
          .from('contractors')
          .select('phone_number')
          .eq('user_id', user.id)
          .maybeSingle()
        if (!contractor?.phone_number) {
          return NextResponse.redirect(new URL('/setup/phone', request.url))
        }
      }

      return response
    }

    export const config = {
      matcher: ['/((?!_next/static|_next/image|favicon.ico|api/onboarding/create-account).*)'],
    }
    ```

    Note: The matcher excludes `api/onboarding/create-account` because the API route does its own `getUser()` check via createSupabaseServer; running middleware on it would also work but adds latency. The matcher excludes `auth/callback` from the redirect-when-authenticated branch (handled inline above) so the OAuth code exchange can complete.

    3. **Create `src/app/(auth)/auth/callback/route.ts`** verbatim from RESEARCH.md Pattern 3:

    ```typescript
    // src/app/(auth)/auth/callback/route.ts
    // Source: RESEARCH.md Pattern 3 — OAuth code exchange.
    import { createServerClient } from '@supabase/ssr'
    import { cookies } from 'next/headers'
    import { NextResponse } from 'next/server'
    import type { NextRequest } from 'next/server'

    export async function GET(request: NextRequest) {
      const { searchParams, origin } = new URL(request.url)
      const code = searchParams.get('code')
      const next = searchParams.get('next') ?? '/setup/phone'

      if (!code) {
        return NextResponse.redirect(`${origin}/auth/login?error=no_code`)
      }

      const cookieStore = await cookies()
      const supabase = createServerClient(
        process.env.NEXT_PUBLIC_SUPABASE_URL!,
        process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
        {
          cookies: {
            getAll: () => cookieStore.getAll(),
            setAll: (cookiesToSet) => {
              cookiesToSet.forEach(({ name, value, options }) =>
                cookieStore.set(name, value, options)
              )
            },
          },
        }
      )

      const { error } = await supabase.auth.exchangeCodeForSession(code)
      if (error) {
        return NextResponse.redirect(`${origin}/auth/login?error=auth_failed`)
      }

      // After OAuth (D-02): redirect to phone capture, which will then redirect to /dashboard
      // once the contractor row has a phone_number.
      return NextResponse.redirect(`${origin}${next}`)
    }
    ```

    3. **Create `src/app/api/onboarding/create-account/route.ts`** verbatim from RESEARCH.md Code Examples (lines 947–999). The route validates with phoneSchema, then uses createServiceRoleClient to insert contractors + contractor_settings + notification_settings rows in a single request:

    ```typescript
    // src/app/api/onboarding/create-account/route.ts
    // Source: RESEARCH.md Code Examples — Account Creation on First Login.
    // Pattern: anon client (server.ts) for auth identity; service role for privileged inserts.
    import { NextResponse } from 'next/server'
    import { createSupabaseServer } from '@/lib/supabase/server'
    import { createServiceRoleClient } from '@/lib/supabase/service'
    import { phoneSchema } from '@/lib/validations/onboarding'

    export async function POST(req: Request) {
      // 1. Verify caller is authenticated (anon client + getUser, NOT getSession)
      const supabase = await createSupabaseServer()
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) {
        return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
      }

      // 2. Validate request body with zod
      const body = phoneSchema.safeParse(await req.json())
      if (!body.success) {
        return NextResponse.json({ error: body.error.issues[0]?.message ?? 'Invalid input' }, { status: 400 })
      }

      // 3. Use service role for write operations (bypasses RLS for atomic multi-row inserts)
      const serviceClient = createServiceRoleClient()

      // 4. Check if a contractor row already exists for this user (idempotency on retry)
      const { data: existing } = await serviceClient
        .from('contractors')
        .select('id, phone_number')
        .eq('user_id', user.id)
        .maybeSingle()

      let contractor_id: string

      if (existing) {
        // Already exists — update phone_number and business_name (handles re-submission)
        const { error: uErr } = await serviceClient
          .from('contractors')
          .update({
            phone_number: body.data.phone_number,
            business_name: body.data.business_name,
            area_code: body.data.area_code,
          })
          .eq('id', existing.id)
        if (uErr) {
          return NextResponse.json({ error: uErr.message }, { status: 500 })
        }
        contractor_id = existing.id
      } else {
        // 5. Insert contractors row
        const { data: contractor, error: cErr } = await serviceClient
          .from('contractors')
          .insert({
            user_id: user.id,
            phone_number: body.data.phone_number,
            business_name: body.data.business_name,
            area_code: body.data.area_code,
          })
          .select()
          .single()
        if (cErr || !contractor) {
          return NextResponse.json({ error: cErr?.message ?? 'Failed to create contractor' }, { status: 500 })
        }
        contractor_id = contractor.id

        // 6. Insert contractor_settings row using DB-level defaults.
        // supported_languages defaults to ['en'] (LANG-02 default; user expands in Plan 06 settings UI).
        // emergency_keywords + business_hours JSONB defaults pre-populated by migration 1 (D-10, D-11).
        const { error: sErr } = await serviceClient
          .from('contractor_settings')
          .insert({
            contractor_id,
            business_name: body.data.business_name,
          })
        if (sErr) {
          // Roll back contractor row to keep state consistent
          await serviceClient.from('contractors').delete().eq('id', contractor_id)
          return NextResponse.json({ error: sErr.message }, { status: 500 })
        }

        // 7. Insert notification_settings row (empty; populated in Plan 06 / Phase 5)
        const { error: nErr } = await serviceClient
          .from('notification_settings')
          .insert({ contractor_id })
        if (nErr) {
          await serviceClient.from('contractor_settings').delete().eq('contractor_id', contractor_id)
          await serviceClient.from('contractors').delete().eq('id', contractor_id)
          return NextResponse.json({ error: nErr.message }, { status: 500 })
        }
      }

      return NextResponse.json({ contractor_id })
    }
    ```
