    ```tsx
    import { usePathname } from 'next/navigation'
    import { PhoneCall, Users, Settings, Bot, Bell, CreditCard, LogOut, Mic } from 'lucide-react'
    import { cn } from '@/lib/utils'
    import { createSupabaseBrowser } from '@/lib/supabase/browser'

    interface SidebarProps {
      businessName?: string | null
      userEmail?: string | null
    }

    const mainNav = [
      { href: '/calls', icon: PhoneCall, label: 'Calls' },
      { href: '/leads', icon: Users, label: 'Leads' },
      { href: '/settings', icon: Settings, label: 'Settings' },
    ]

    const settingsSubNav = [
      { href: '/settings/agent', icon: Bot, label: 'Agent Config' },
      { href: '/settings/notifications', icon: Bell, label: 'Notifications' },
      { href: '/settings/account', icon: CreditCard, label: 'Account & Billing' },

    export function Sidebar({ businessName, userEmail }: SidebarProps) {

      const onSettings = pathname.startsWith('/settings')

        const supabase = createSupabaseBrowser()
        await supabase.auth.signOut()

      return (
        <aside
          className="w-60 h-screen fixed bg-[#0F172A] border-r border-[#334155] flex-col hidden md:flex"

          {/* Logo area */}

          {/* Nav */}
          <nav className="flex-1 overflow-y-auto py-6">
            <p className="text-sm font-semibold text-[#94A3B8] tracking-wider px-4 mb-2">NAVIGATION</p>

              const isActive =
                href === '/settings'
                  ? pathname.startsWith('/settings')

              return (
                <Link
                  key={href}
                  href={href}
                  className={cn(
                    'flex items-center gap-3 py-3 rounded-lg mx-2 min-h-[48px]',

                      ? 'text-[#F8FAFC] bg-[#1E293B] border-l-4 border-[#F97316] pl-[12px]'

                  <span className={isActive ? 'font-semibold' : 'font-normal'}>{label}</span>
                </Link>
              )
            })}

                {settingsSubNav.map(({ href, icon: Icon, label }) => {

                    <Link
                      key={href}
                      href={href}

                        'flex items-center gap-3 py-2 rounded-lg mx-1 min-h-[40px] text-sm',
    ```

    ```tsx
    // Source: RESEARCH.md Pattern 6 + PATTERNS.md DashboardLayout.
    import { redirect } from 'next/navigation'
    import { createSupabaseServer } from '@/lib/supabase/server'
    import { Sidebar } from '@/components/dashboard/Sidebar'
    import { SetupBanner } from '@/components/dashboard/SetupBanner'
    import { BottomTabBar } from '@/components/dashboard/BottomTabBar'

    export default async function DashboardLayout({
      children,
    }: {
      children: React.ReactNode
    }) {
      const supabase = await createSupabaseServer()
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) redirect('/auth/login')

      const { data: contractor } = await supabase
        .from('contractors')
        .select('id, business_name, phone_number')

        <div className="min-h-screen bg-[#1E293B]">

          <main className="md:ml-60 min-h-screen pb-14 md:pb-0">

              {children}
    ```

    ```tsx
    // src/app/(dashboard)/settings/agent/page.tsx — Server Component
    import { redirect } from 'next/navigation'
    import { createSupabaseServer } from '@/lib/supabase/server'
    import { createServiceRoleClient } from '@/lib/supabase/service'
    import { LanguageSettingsForm } from './_components/LanguageSettingsForm'
    import type { Language } from '@/types/database'

    export const metadata = { title: 'Agent Config — VoiceDesk' }

    export default async function AgentConfigPage() {
      const supabase = await createSupabaseServer()
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) redirect('/auth/login')

      // Look up the contractor for this user
      const { data: contractor } = await supabase
        .from('contractors')
        .select('id')
        .eq('user_id', user.id)
        .maybeSingle()
      if (!contractor) redirect('/setup/phone')

      // Read current supported_languages via service role (consistent with mutation pattern;
      // RLS would also allow this read for the owning user, but we use service role to mirror
      // the write path used by the updateLanguages server action — see src/app/actions/settings.ts)
      const serviceClient = createServiceRoleClient()
      const { data: settings } = await serviceClient
        .from('contractor_settings')
        .select('supported_languages')
        .eq('contractor_id', contractor.id)
        .maybeSingle()

      const current: Language[] = (settings?.supported_languages as Language[] | null) ?? ['en']

      return (
        <div>
          <div className="mb-6">
            <h1 className="text-xl font-semibold text-[#F8FAFC] mb-1">Agent Configuration</h1>
            <p className="text-sm text-[#94A3B8]">
              Choose the languages your AI receptionist will speak (English, Spanish, Haitian Creole). Business name,
              hours, and after-hours message ship in Phase 6.
            </p>
            <div className="border-b border-[#334155] mt-6" />
          </div>
          <LanguageSettingsForm current={current} />
        </div>
      )
    ```

    11. **Create `src/app/actions/settings.ts`** with the `updateLanguages` server action. The action: (a) calls `createSupabaseServer().auth.getUser()` for authorization (NOT getSession — RESEARCH.md Pitfall 2), (b) validates input with `languageSchema` from `@/lib/validations/onboarding` (Plan 03), (c) updates `contractor_settings.supported_languages` via `createServiceRoleClient` for the contractor owned by the authenticated user, and (d) calls `revalidatePath('/settings/agent')` so the page re-reads the updated value on next render.

    ```typescript
    // src/app/actions/settings.ts
    // Source: D-03 server-action pattern from CONTEXT.md; Plan 03 languageSchema; service role for mutations.
    'use server'
    import { revalidatePath } from 'next/cache'
    import { createSupabaseServer } from '@/lib/supabase/server'
    import { createServiceRoleClient } from '@/lib/supabase/service'
    import { languageSchema, type LanguageFormData } from '@/lib/validations/onboarding'

    type UpdateResult = { ok: true } | { ok: false; error: string }

    export async function updateLanguages(input: LanguageFormData): Promise<UpdateResult> {
      // 1. Authorization: getUser() validates JWT against Supabase Auth (RESEARCH.md Pitfall 2)
      const supabase = await createSupabaseServer()
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) return { ok: false, error: 'Not signed in.' }

      // 2. Validate input against Plan 03 zod schema (LANG-02 floor: ['en'|'es'|'ht'][], min 1)
      const parsed = languageSchema.safeParse(input)
      if (!parsed.success) {
        return { ok: false, error: parsed.error.issues[0]?.message ?? 'Invalid language selection.' }
      }

      // 3. Find this user's contractor row
      const serviceClient = createServiceRoleClient()
      const { data: contractor, error: cErr } = await serviceClient
        .from('contractors')
        .select('id')
        .eq('user_id', user.id)
        .maybeSingle()
      if (cErr || !contractor) return { ok: false, error: 'Contractor not found for this user.' }

      // 4. Update supported_languages on contractor_settings
      const { error: uErr } = await serviceClient
        .from('contractor_settings')
        .update({ supported_languages: parsed.data.supported_languages })
        .eq('contractor_id', contractor.id)
      if (uErr) return { ok: false, error: uErr.message }

      revalidatePath('/settings/agent')
      return { ok: true }
    }
    ```
