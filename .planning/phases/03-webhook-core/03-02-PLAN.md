// src/lib/retell/webhook-types.ts
// Typed interfaces for Retell webhook payloads.
// Derived from PhoneCallResponse in retell-sdk/resources/call.d.ts (verified 2026-05-08).
// CRITICAL: call.call_analysis?.call_summary (NOT call.summary or call.analysis.summary)
// CRITICAL: call.duration_ms is milliseconds — divide by 1000 for calls.duration_seconds
// CRITICAL: store call.transcript_object in calls.transcript JSONB (NOT call.transcript string)
import 'server-only'

export interface RetellCallPayload {
  call_id: string
  agent_id: string
  from_number: string          // E.164 caller phone → calls.caller_phone
  to_number: string            // VoiceDesk number → calls.twilio_number
  direction: 'inbound' | 'outbound'
  call_status: string
  start_timestamp?: number     // ms epoch → new Date(start_timestamp) → calls.started_at
  end_timestamp?: number       // ms epoch → new Date(end_timestamp) → calls.ended_at
  duration_ms?: number         // milliseconds → Math.round(duration_ms / 1000) → calls.duration_seconds
  recording_url?: string       // D-16: store as-is in calls.recording_url
  transcript?: string          // plain speaker-labeled string — do NOT store in JSONB column
  transcript_object?: Array<{  // structured array → calls.transcript (JSONB)
    role: 'agent' | 'user' | 'transfer_target'
    content: string
    words: Array<{ word?: string; start?: number; end?: number }>
  }>
  call_analysis?: {
    call_summary?: string      // D-09: → calls.summary TEXT (NOT "summary", NOT "call_analysis.summary")
    call_successful?: boolean
    in_voicemail?: boolean
    user_sentiment?: 'Negative' | 'Positive' | 'Neutral' | 'Unknown'
    custom_analysis_data?: unknown
  }
  disconnection_reason?: string
  metadata?: unknown
}

export interface RetellWebhookEvent {
  event: 'call_started' | 'call_ended' | 'call_analyzed' | 'transcript_updated' | string
  call: RetellCallPayload
}
```

// src/lib/classification/emergency-keywords.ts
// EMRG-02: Hard-coded emergency triggers that CANNOT be overridden by contractor configuration.
// Checked BEFORE any LLM call in the classification pipeline.
// D-06: If matched, intent='emergency' is final — LLM cannot override.
// Gas smell and CO alarm are non-configurable by design — safety requirement.
import 'server-only'

export const HARD_EMERGENCY_KEYWORDS: Record<string, readonly string[]> = {
  en: [
    'gas smell',
    'gas leak',
    'co alarm',
    'carbon monoxide',
    'co detector',
    'carbon monoxide alarm',
  ],
  es: [
    'olor a gas',
    'fuga de gas',
    'alarma de co',
    'monóxido de carbono',
    'detector de co',
    'alarma de monóxido',
  ],
} as const

/**
 * Returns true if the transcript contains any hard-coded emergency keyword (EN or ES).
 * Case-insensitive. These keywords cannot be disabled by contractor configuration (EMRG-02).
 */
export function hasHardEmergencyKeyword(transcriptText: string): boolean {
  const lower = transcriptText.toLowerCase()
  return Object.values(HARD_EMERGENCY_KEYWORDS).flat().some(kw => lower.includes(kw))
}

// src/lib/after-hours/detect.ts
// Pure timestamp comparison for after-hours detection.
// D-11: no LLM — compare callStartAt against contractor business_hours JSONB.
// Uses date-fns-tz toZonedTime() for DST-safe arithmetic (America/New_York handles EDT/EST).
// timezone: read from contractors.timezone column (default 'America/New_York') — NOT contractor_settings.
import 'server-only'
import { toZonedTime } from 'date-fns-tz'
import { getDay, getHours, getMinutes } from 'date-fns'

export interface BusinessHours {
  [day: string]: { open: string; close: string; enabled: boolean }
}

// Maps Date.getDay() index (0=Sunday) to business_hours JSONB key names
const DAY_NAMES = ['sunday', 'monday', 'tuesday', 'wednesday', 'thursday', 'friday', 'saturday'] as const

/**
 * Returns true if the call started outside the contractor's configured business hours.
 * callStartAt: UTC Date (converted from Retell start_timestamp ms epoch)
 * businessHours: contractor_settings.business_hours JSONB — { monday: { open: 'HH:MM', close: 'HH:MM', enabled: boolean }, ... }
 * timezone: contractors.timezone TEXT (e.g. 'America/New_York') — DST-aware via date-fns-tz
 */
export function detectAfterHours(
  callStartAt: Date,
  businessHours: BusinessHours,
  timezone: string
): boolean {
  const zonedDate = toZonedTime(callStartAt, timezone)
  const dayName = DAY_NAMES[getDay(zonedDate)]
  const dayConfig = businessHours[dayName]

  // Day not configured or disabled → always after-hours
  if (!dayConfig?.enabled) return true

  const [openH, openM] = dayConfig.open.split(':').map(Number)
  const [closeH, closeM] = dayConfig.close.split(':').map(Number)

  const currentMinutes = getHours(zonedDate) * 60 + getMinutes(zonedDate)
  const openMinutes = openH * 60 + openM
  const closeMinutes = closeH * 60 + closeM

  // Exactly at open time is within hours; exactly at close is after-hours
  return currentMinutes < openMinutes || currentMinutes >= closeMinutes
}
```
