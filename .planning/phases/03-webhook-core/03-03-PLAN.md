```typescript
// src/lib/classification/classify-intent.ts
// GPT-4o-mini intent classification with Zod-enforced structured output.
// D-07/D-08: called after hard-code emergency check; result can be overridden by D-06 in Inngest.
// D-12: Intent enum values MUST match Phase 1 schema CHECK constraint exactly:
//   'lead' | 'emergency' | 'info' | 'no_action' — NOT 'general_inquiry', NOT 'unknown'
// D-13: isShortCall=true is passed in prompt context for calls < 15 seconds.
import 'server-only'
import { zodResponseFormat } from 'openai/helpers/zod'
import { z } from 'zod'
import { getOpenAIClient } from '@/lib/openai/client'

// D-12: enum values must match Phase 1 schema CHECK constraint exactly
export const ClassificationSchema = z.object({
  intent: z.enum(['lead', 'emergency', 'info', 'no_action']),
  detected_trade: z.enum(['hvac', 'plumbing', 'electrical', 'other']).optional(),
  confidence: z.enum(['high', 'medium', 'low']),
})

export type ClassificationResult = z.infer<typeof ClassificationSchema>

function buildClassificationPrompt(isShortCall: boolean): string {
  const shortCallNote = isShortCall
    ? '\n\nNOTE: This call was very short (under 15 seconds). If the caller did not express a clear need, classify as no_action.'
    : ''
  return `You are classifying inbound calls to HVAC, plumbing, and electrical contractors.

Classify the call into exactly one intent:
- "lead": caller has a service need (repair, installation, maintenance, estimate, general inquiry with clear interest)
- "emergency": caller describes an urgent safety or damage situation happening now (flooding, gas issue confirmed NOW, no power, sparking)
- "info": caller asking questions without a clear service need, or general inquiry without urgency
- "no_action": caller hung up, wrong number, spam, or call too short to determine intent

IMPORTANT: Gas smell and CO alarm are handled separately — if you see these in the transcript, still classify normally; the calling system will override with 'emergency' anyway.

Also identify the trade type if determinable: hvac, plumbing, electrical, or other.${shortCallNote}`
}

export async function classifyIntent(
  transcriptText: string,
  isShortCall: boolean
): Promise<ClassificationResult> {
  const client = getOpenAIClient()
  const completion = await client.chat.completions.parse({
    model: 'gpt-4o-mini',
    messages: [
      { role: 'system', content: buildClassificationPrompt(isShortCall) },
      { role: 'user', content: transcriptText || '[empty transcript]' },
    ],
    response_format: zodResponseFormat(ClassificationSchema, 'classification'),
  })
  const result = completion.choices[0]?.message?.parsed
  if (!result) throw new Error('[classify-intent] OpenAI returned no parsed result')
  return result
}
```

```typescript
// src/lib/classification/extract-trade.ts
// GPT-4o trade-specific data extraction — second LLM call in the D-08 pipeline.
// D-10: trade-specific Zod schemas per contractor's trade_type.
// Uses gpt-4o (not gpt-4o-mini) for better structured extraction accuracy.
// Returns null for no_action intent to avoid wasted API calls (D-13).
import 'server-only'
import { zodResponseFormat } from 'openai/helpers/zod'
import { z } from 'zod'
import { getOpenAIClient } from '@/lib/openai/client'

// D-10: Per-trade extraction schemas
const HvacSchema = z.object({
  system_age: z.string().optional(),
  issue_type: z.enum(['heating', 'cooling', 'other']),
  urgency: z.enum(['normal', 'urgent', 'emergency']),
})

const PlumbingSchema = z.object({
  leak_type: z.enum(['active', 'slow', 'none']),
  repair_type: z.string().optional(),
  urgency: z.enum(['normal', 'urgent', 'emergency']),
})

const ElectricalSchema = z.object({
  safety_issue: z.boolean(),
  repair_type: z.string().optional(),
  urgency: z.enum(['normal', 'urgent', 'emergency']),
})

// Fallback schema for multi-trade or unknown trade types
const GeneralSchema = z.object({
  issue_description: z.string().optional(),
  urgency: z.enum(['normal', 'urgent', 'emergency']),
})

export type TradeExtractionResult =
  | z.infer<typeof HvacSchema>
  | z.infer<typeof PlumbingSchema>
  | z.infer<typeof ElectricalSchema>
  | z.infer<typeof GeneralSchema>

function getSchemaForTrade(trade?: string): { schema: z.ZodObject<z.ZodRawShape>; name: string } {
  switch (trade) {
    case 'hvac': return { schema: HvacSchema, name: 'hvac_extraction' }
    case 'plumbing': return { schema: PlumbingSchema, name: 'plumbing_extraction' }
    case 'electrical': return { schema: ElectricalSchema, name: 'electrical_extraction' }
    default: return { schema: GeneralSchema, name: 'general_extraction' }
  }
}

function buildExtractionPrompt(trade?: string): string {
  return `Extract structured information from this ${trade ?? 'general'} contractor call transcript.
Fill in only what was explicitly stated by the caller. Use your best judgment for urgency.
Urgency guide: emergency = happening now + safety risk, urgent = active damage or no functionality, normal = maintenance or future repair.`
}

/**
 * Extracts trade-specific structured data from the call transcript.
 * Returns null if the transcript is empty or not worth extracting (e.g., no_action calls).
 * detectedTrade: from classification step or contractor.trade_type
 */
export async function extractTradeData(
  transcriptText: string,
  detectedTrade?: string
): Promise<TradeExtractionResult | null> {
  if (!transcriptText?.trim()) return null

  const { schema, name } = getSchemaForTrade(detectedTrade)
  const client = getOpenAIClient()

  const completion = await client.chat.completions.parse({
    model: 'gpt-4o',
    messages: [
      { role: 'system', content: buildExtractionPrompt(detectedTrade) },
      { role: 'user', content: transcriptText },
    ],
    response_format: zodResponseFormat(schema, name),
  })

  const result = completion.choices[0]?.message?.parsed
  if (!result) {
    console.error('[extract-trade] OpenAI returned no parsed result for trade:', detectedTrade)
    return null
  }
  return result as TradeExtractionResult
}
```

```typescript
// src/app/api/webhooks/retell/route.ts
// Retell AI webhook handler — single route dispatching call_ended + call_analyzed events.
// D-01: Single route, dispatches by event.event type.
// CRITICAL: verify() called BEFORE parsing payload (req.text() not req.json()).
// CRITICAL: Returns HTTP 200 immediately — all downstream processing in after() (CLAUDE.md rule 2).
// Security: T-03-01 — verify() rejects forged webhooks with 401 before any payload access.
// Security: T-03-02 — contractor lookup by agent_id, never by contractor_id from payload.
import 'server-only'
import { after } from 'next/server'
import { verify } from 'retell-sdk'
import { processCallEnded, processCallAnalyzed } from './handlers'
import type { RetellWebhookEvent } from '@/lib/retell/webhook-types'

// Vercel segment config — after() budget must cover: 1 Supabase write + 1 Inngest send
export const maxDuration = 30
export const runtime = 'nodejs'

export async function POST(req: Request) {
  // MUST call req.text() BEFORE req.json() — verify() needs the raw string body
  const rawBody = await req.text()
  const signature = req.headers.get('x-retell-signature') ?? ''

  // T-03-01: Verify HMAC signature before trusting any payload content
  const isValid = await verify(rawBody, process.env.RETELL_API_KEY!, signature)
  if (!isValid) {
    console.error('[webhook/retell] invalid signature — rejecting request')
    return new Response('Unauthorized', { status: 401 })
  }

  const event = JSON.parse(rawBody) as RetellWebhookEvent

  // Return 200 immediately — critical architecture rule (CLAUDE.md)
  // All downstream processing deferred to after()
  after(async () => {
    try {
      if (event.event === 'call_ended') {
        await processCallEnded(event.call)
      } else if (event.event === 'call_analyzed') {
        await processCallAnalyzed(event.call)
      }
      // call_started, transcript_updated, transfer_* — no-op (not subscribed to these events)
    } catch (err) {
      console.error('[webhook/retell] after() processing error:', err instanceof Error ? err.message : err)
    }
  })

  return new Response('OK', { status: 200 })
}
```

```typescript
// src/app/api/webhooks/retell/handlers.ts
// call_ended: creates skeleton calls row (analysis_status='analyzing')
// call_analyzed: enriches row with transcript + summary, fires Inngest classification job
// T-03-02: contractor_id is NEVER taken from the webhook payload — always looked up by agent_id server-side.
// D-05: Both handlers use UPSERT on retell_call_id — idempotent on duplicate webhook delivery.
import 'server-only'
import { createServiceRoleClient } from '@/lib/supabase/service'
import { inngest } from '@/inngest/client'
import { detectAfterHours } from '@/lib/after-hours/detect'
import type { RetellCallPayload } from '@/lib/retell/webhook-types'
import type { BusinessHours } from '@/lib/after-hours/detect'

/**
 * call_ended: creates skeleton calls row.
 * Sets status='completed', analysis_status='analyzing', is_after_hours.
 * contractor_id is resolved server-side from call.agent_id (T-03-02 — never from payload).
 * D-05: UPSERT on retell_call_id handles duplicate webhook deliveries idempotently.
 */
export async function processCallEnded(call: RetellCallPayload): Promise<void> {
  const db = createServiceRoleClient()

  // T-03-02: Look up contractor by retell_agent_id from webhook payload
  // NEVER accept contractor_id directly from the payload — attacker could supply wrong ID
  const { data: contractor, error: lookupError } = await db
    .from('contractors')
    .select('id, timezone, contractor_settings(business_hours)')
    .eq('retell_agent_id', call.agent_id)
    .maybeSingle()

  if (lookupError) {
    console.error('[webhook/call_ended] contractor lookup error:', lookupError.message)
    return
  }
  if (!contractor) {
    console.error('[webhook/call_ended] no contractor found for agent_id:', call.agent_id)
    return // Unknown agent — not our event; return silently
  }

  // D-11: After-hours detection — pure timestamp comparison, no LLM
  const settings = contractor.contractor_settings as { business_hours: BusinessHours } | null
  const isAfterHours = settings?.business_hours && call.start_timestamp
    ? detectAfterHours(
        new Date(call.start_timestamp),
        settings.business_hours,
        contractor.timezone ?? 'America/New_York'
      )
    : false

  // D-05: UPSERT on retell_call_id — idempotent; duplicate call_ended → same row updated
  const { error: upsertError } = await db.from('calls').upsert(
    {
      contractor_id: contractor.id,
      retell_call_id: call.call_id,
      caller_phone: call.from_number ?? null,
      twilio_number: call.to_number ?? null,
      status: 'completed',
      analysis_status: 'analyzing',  // D-02/D-03: skeleton row state
      started_at: call.start_timestamp ? new Date(call.start_timestamp).toISOString() : null,
      ended_at: call.end_timestamp ? new Date(call.end_timestamp).toISOString() : null,
      duration_seconds: call.duration_ms ? Math.round(call.duration_ms / 1000) : null,  // duration_ms NOT duration_seconds
      recording_url: call.recording_url ?? null,  // D-16: store as-is
      is_after_hours: isAfterHours,
    },
    { onConflict: 'retell_call_id' }  // D-05: idempotency
  )

  if (upsertError) {
    console.error('[webhook/call_ended] calls upsert failed:', upsertError.message)
  }
}

/**
 * call_analyzed: enriches the calls row with transcript + summary, queues Inngest classification.
 * ALWAYS persists transcript before firing Inngest — never lose the transcript on OpenAI failure.
 * D-09: call_analysis.call_summary → calls.summary (NOT call.summary or call.analysis.summary)
 * D-05: UPSERT handles race where call_analyzed arrives before call_ended.
 */
export async function processCallAnalyzed(call: RetellCallPayload): Promise<void> {
  const db = createServiceRoleClient()

  // Always persist transcript + summary FIRST — even if Inngest send fails later
  const { data: callRow, error: upsertError } = await db
    .from('calls')
    .upsert(
      {
        retell_call_id: call.call_id,
        // D-09: call_analysis.call_summary is the field (NOT call.summary)
        summary: call.call_analysis?.call_summary ?? null,
        // Store transcript_object (structured array) in JSONB — NOT the plain transcript string
        transcript: call.transcript_object ?? null,
        recording_url: call.recording_url ?? null,
        // analysis_status stays 'analyzing' — Inngest classify job will set 'complete'
      },
      { onConflict: 'retell_call_id' }
    )
    .select('id, contractor_id, duration_ms')
    .maybeSingle()

  if (upsertError) {
    console.error('[webhook/call_analyzed] transcript upsert failed:', upsertError.message)
    return
  }
  if (!callRow) {
```
