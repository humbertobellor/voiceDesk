```typescript
import 'server-only'
import { createServiceRoleClient } from '@/lib/supabase/service'
import type { NotificationChannel } from '@/types/database'

export interface WriteNotificationLogParams {
  contractorId: string
  callId: string | null
  channel: NotificationChannel
  recipient: string
  message: string
}

// D-12: Write row BEFORE send (status='pending') — ensures log entry exists even if process crashes.
// Returns the new notification_log row UUID.
export async function writeNotificationLog(params: WriteNotificationLogParams): Promise<string> {
  const db = createServiceRoleClient()
  const { data, error } = await db
    .from('notification_log')
    .insert({
      contractor_id: params.contractorId,
      call_id: params.callId,
      channel: params.channel,
      recipient: params.recipient,
      message: params.message,
      status: 'pending',
    })
    .select('id')
    .single()

  if (error || !data) {
    throw new Error(`[notification_log] insert failed: ${error?.message ?? 'no data returned'}`)
  }
  return data.id as string
}

// D-12: Update row AFTER send resolves — set status='sent'/'failed', sent_at, error.
export async function updateNotificationLog(
  logId: string,
  update: { status: 'sent' | 'failed'; sentAt?: string; error?: string }
): Promise<void> {
  const db = createServiceRoleClient()
  const { error } = await db
    .from('notification_log')
    .update({
      status: update.status,
      sent_at: update.sentAt ?? (update.status === 'sent' ? new Date().toISOString() : null),
      error: update.error ?? null,
    })
    .eq('id', logId)

  if (error) {
    // Non-fatal — log the failure but do not throw (avoids cascading retry on a logging error)
    console.error(`[notification_log] update failed for log ${logId}:`, error.message)
  }
}
```

```typescript
import 'server-only'
import twilio from 'twilio'

export interface SendSmsParams {
  to: string       // E.164 format: +1XXXXXXXXXX
  body: string
  contractorId: string
  callId: string | null
}

export interface SendSmsResult {
  sid: string
  status: string
}

// NOTIF-01: Send SMS to contractor phone number via Twilio.
// EMRG-03: Emergency SMS must include "EMERGENCY" in body (enforced by caller in dispatch function).
export async function sendSms(params: SendSmsParams): Promise<SendSmsResult> {
  const accountSid = process.env.TWILIO_ACCOUNT_SID
  const authToken = process.env.TWILIO_AUTH_TOKEN
  const fromNumber = process.env.TWILIO_FROM_NUMBER

  if (!accountSid) throw new Error('[sendSms] TWILIO_ACCOUNT_SID is not set')
  if (!authToken) throw new Error('[sendSms] TWILIO_AUTH_TOKEN is not set')
  if (!fromNumber) throw new Error('[sendSms] TWILIO_FROM_NUMBER is not set')

  const client = twilio(accountSid, authToken)
  const message = await client.messages.create({
    to: params.to,
    from: fromNumber,
    body: params.body,
  })

  return { sid: message.sid, status: message.status }
}
```

```typescript
import 'server-only'
import twilio from 'twilio'

export interface SendWhatsAppParams {
  to: string            // E.164 format: +1XXXXXXXXXX (no whatsapp: prefix — added internally)
  templateType: 'new_lead' | 'emergency'
  callerPhone: string   // {{1}} in both templates
  serviceRequested: string  // {{2}} in both templates
  bookingStatus?: string    // {{3}} — new_lead template only
  contractorId: string
  callId: string | null
}

export interface SendWhatsAppResult {
  sid: string
  status: string
}

// D-16: WhatsApp Content SIDs stored as TypeScript constants for v1 — NOT env vars.
// Replace the HX placeholder values with the real Content SIDs from Twilio Console ->
// Messaging -> Content Template Builder after WABA template approval (voicedesk_new_lead,
// voicedesk_emergency). Do NOT use env var reads here — D-16 requires pure constants.
export const WHATSAPP_CONTENT_SIDS = {
  // voicedesk_new_lead: "New call from {{1}}\nService: {{2}}\nBooking: {{3}}"
  new_lead: 'HXplaceholder_new_lead',
  // voicedesk_emergency: "EMERGENCY: {{1}} called about {{2}}. Call back immediately."
  emergency: 'HXplaceholder_emergency',
} as const

// D-14 + D-15: Send WhatsApp template message.
// NOTIF-02: Pre-approved WABA template required — Content SID maps to approved template.
// EMRG-03: Emergency template carries "EMERGENCY" in the pre-approved template body.
export async function sendWhatsApp(params: SendWhatsAppParams): Promise<SendWhatsAppResult> {
  const accountSid = process.env.TWILIO_ACCOUNT_SID
  const authToken = process.env.TWILIO_AUTH_TOKEN
  const fromNumber = process.env.TWILIO_WHATSAPP_FROM

  if (!accountSid) throw new Error('[sendWhatsApp] TWILIO_ACCOUNT_SID is not set')
  if (!authToken) throw new Error('[sendWhatsApp] TWILIO_AUTH_TOKEN is not set')
  if (!fromNumber) throw new Error('[sendWhatsApp] TWILIO_WHATSAPP_FROM is not set')

  const contentSid = WHATSAPP_CONTENT_SIDS[params.templateType]

  // D-14: new_lead template variables: {{1}}=callerPhone, {{2}}=serviceRequested, {{3}}=bookingStatus
  // D-15: emergency template variables: {{1}}=callerPhone, {{2}}=serviceRequested
  const contentVariables: Record<string, string> =
    params.templateType === 'new_lead'
      ? {
          '1': params.callerPhone,
          '2': params.serviceRequested,
          '3': params.bookingStatus ?? 'No appointment booked',
        }
      : {
          '1': params.callerPhone,
          '2': params.serviceRequested,
        }

  const client = twilio(accountSid, authToken)
  const message = await client.messages.create({
    to: `whatsapp:${params.to}`,
    from: fromNumber,  // D-17: TWILIO_WHATSAPP_FROM is already 'whatsapp:+1XXXXXXXXXX'
    contentSid,
    contentVariables: JSON.stringify(contentVariables),
  })

  return { sid: message.sid, status: message.status }
}
```

```typescript
import 'server-only'
import { Bot } from 'grammy'

export interface SendTelegramParams {
  chatId: string   // Telegram chat_id stored in notification_settings.telegram_chat_id
  message: string  // Plain text — no template approval needed for Telegram
  contractorId: string
  callId: string | null
}

export interface SendTelegramResult {
  messageId: number
}

// D-18: Single @VoiceDeskBot for the entire platform. One TELEGRAM_BOT_TOKEN env var.
// D-22: grammY webhook mode (not polling) — Bot instance is created per-request, not persistent.
// NOTIF-03: Sends plain text notification to contractor's linked Telegram account.
export async function sendTelegram(params: SendTelegramParams): Promise<SendTelegramResult> {
  const botToken = process.env.TELEGRAM_BOT_TOKEN
  if (!botToken) throw new Error('[sendTelegram] TELEGRAM_BOT_TOKEN is not set')

  const bot = new Bot(botToken)
  const result = await bot.api.sendMessage(params.chatId, params.message)

  return { messageId: result.message_id }
}
```
