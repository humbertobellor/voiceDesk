    ```typescript
    /**
     * Mutation that PATCHes /api/notifications/{id}/mark-as-read to mark a single
     * notification as read for the authenticated contractor.
     *
     * Mirrors the useMarkAllRead pattern but operates on a single notification.
     * Applies optimistic update to the notifications array (find by id, set is_read=true).
     * Invalidates both the notifications list and unread count on settle.
     *
     * @param notificationId - The ID of the notification to mark as read
     * @returns UseMutationResult.
     */
    export function useMarkNotificationAsRead() {
      const queryClient = useQueryClient()
      const { contractorId } = useContractor()

      return useMutation({
        mutationFn: async (notificationId: string): Promise<void> => {
          const response = await fetch(`/api/notifications/${notificationId}/mark-as-read`, {
            method: 'PATCH',
          })
          if (!response.ok) {
            const text = await response.text()
            throw new Error(`Failed to mark notification as read: ${text}`)
          }
        },
        onMutate: (notificationId: string) => {
          // Optimistic update: find the notification in the cache and set is_read=true.
          if (contractorId) {
            queryClient.setQueryData(
              ['notifications', contractorId],
              (old: AppNotification[] | undefined) => {
                if (!old) return old
                return old.map((n) =>
                  n.id === notificationId ? { ...n, is_read: true, read_at: new Date().toISOString() } : n
                )
              }
            )
            // Also decrement unread count by 1 (if the notification was unread).
            queryClient.setQueryData(
              ['notifications-unread', contractorId],
              (old: number | undefined) => (old && old > 0 ? old - 1 : 0)
            )
          }
        },
        onSettled: () => {
          queryClient.invalidateQueries({ queryKey: ['notifications', contractorId] })
          queryClient.invalidateQueries({ queryKey: ['notifications-unread', contractorId] })
        },
      })
    }
    ```

    Key points:
    - mutationFn accepts notificationId as parameter (unlike mark-all-read which takes no params)
    - fetch uses template literal: /api/notifications/${notificationId}/mark-as-read
    - onMutate updates the notifications array: find by id, set is_read=true and read_at
    - onMutate also decrements unread count by 1 (optimistic)
    - onSettled invalidates both queries to refresh on success/error
    - Use AppNotification type from imports (already imported at top)

    Do NOT add a new import — AppNotification, queryClient, contractorId are already available.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → PATCH /api/notifications/[id]/mark-as-read | Single-notification mark-as-read endpoint — contractor_id from session, id from URL param |
| Browser → React Query cache | Optimistic update before server confirms; rollback on error |
| Browser → Supabase Realtime | Badge count updates reflect server state |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-17-02-01 | Tampering | Notification ID from URL param | mitigate | API validates contractor_id from session matches notification owner (IDOR check: SELECT by id AND contractor_id). If mismatch → 403 Forbidden. |
| T-17-02-02 | Elevation of Privilege | Client-side is_read flag | mitigate | API source-of-truth: is_read only set server-side in UPDATE statement. Optimistic update in client is visual-only; server state is authoritative. |
| T-17-02-03 | IDOR | Double-submit race | mitigate | Idempotent UPDATE: notification already marked is_read is excluded on retry. No duplicate writes or side effects. |
| T-17-02-04 | Information Disclosure | Optimistic update before settle | mitigate | Optimistic update limited to: is_read flag and read_at timestamp. No sensitive fields exposed. On error, onSettled invalidates cache, rolling back optimistic state. |
</threat_model>
