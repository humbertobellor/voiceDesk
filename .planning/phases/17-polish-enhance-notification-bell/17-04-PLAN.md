    Structure (React Testing Library + renderHook pattern):

    ```typescript
    import { describe, it, expect, beforeEach, vi } from 'vitest'
    import { renderHook, act } from '@testing-library/react'
    import { QueryClient, QueryClientProvider } from '@tanstack/react-query'
    import { useMarkNotificationAsRead } from '@/lib/query/client'
    import { useContractor } from '@/contexts/ContractorContext'

    // Mock dependencies
    vi.mock('@/contexts/ContractorContext', () => ({
      useContractor: vi.fn(() => ({ contractorId: 'test-contractor-id' }))
    }))

    global.fetch = vi.fn()

    describe('useMarkNotificationAsRead', () => {
      let queryClient: QueryClient

      beforeEach(() => {
        queryClient = new QueryClient({
          defaultOptions: {
            queries: { retry: false },
            mutations: { retry: false }
          }
        })
        vi.clearAllMocks()
      })

      it('should call fetch with correct endpoint and method', async () => {
        const wrapper = ({ children }: { children: React.ReactNode }) => (
          <QueryClientProvider client={queryClient}>{children}</QueryClientProvider>
        )

        const { result } = renderHook(() => useMarkNotificationAsRead(), { wrapper })

        await act(async () => {
          ;(global.fetch as any).mockResolvedValueOnce(
            new Response(null, { status: 200 })
          )
          result.current.mutate('notification-id-123')
        })

        expect(global.fetch).toHaveBeenCalledWith(
          '/api/notifications/notification-id-123/mark-as-read',
          { method: 'PATCH' }
        )
      })

      it('should apply optimistic update to notifications array', async () => {
        // Setup: queryClient has initial notifications cache
        queryClient.setQueryData(
          ['notifications', 'test-contractor-id'],
          [
            { id: 'notif-1', is_read: false, created_at: new Date().toISOString() },
            { id: 'notif-2', is_read: true, created_at: new Date().toISOString() }
          ]
        )

        const wrapper = ({ children }: { children: React.ReactNode }) => (
          <QueryClientProvider client={queryClient}>{children}</QueryClientProvider>
        )

        const { result } = renderHook(() => useMarkNotificationAsRead(), { wrapper })

        // Call mutate and check cache immediately (optimistic)
        act(() => {
          result.current.mutate('notif-1')
        })

        const cached = queryClient.getQueryData(['notifications', 'test-contractor-id']) as any[]
        expect(cached[0].is_read).toBe(true) // Optimistic update applied
      })

      it('should decrement unread count in optimistic update', async () => {
        queryClient.setQueryData(['notifications-unread', 'test-contractor-id'], 3)

        const wrapper = ({ children }: { children: React.ReactNode }) => (
          <QueryClientProvider client={queryClient}>{children}</QueryClientProvider>
        )

        const { result } = renderHook(() => useMarkNotificationAsRead(), { wrapper })

        act(() => {
          result.current.mutate('notif-1')
        })

        const unreadCount = queryClient.getQueryData(['notifications-unread', 'test-contractor-id'])
        expect(unreadCount).toBe(2) // Decremented by 1
      })

      it('should handle errors and set isError', async () => {
        const wrapper = ({ children }: { children: React.ReactNode }) => (
          <QueryClientProvider client={queryClient}>{children}</QueryClientProvider>
        )

        const { result } = renderHook(() => useMarkNotificationAsRead(), { wrapper })

        await act(async () => {
          ;(global.fetch as any).mockResolvedValueOnce(
            new Response(null, { status: 500 })
          )
          result.current.mutate('notif-1')
        })

        expect(result.current.isError).toBe(true)
      })

      it('should set isPending while mutation is in-flight', async () => {
        ;(global.fetch as any).mockImplementationOnce(
          () => new Promise(resolve =>
            setTimeout(() => resolve(new Response(null, { status: 200 })), 100)
          )
        )

        const wrapper = ({ children }: { children: React.ReactNode }) => (
          <QueryClientProvider client={queryClient}>{children}</QueryClientProvider>
        )

        const { result } = renderHook(() => useMarkNotificationAsRead(), { wrapper })

        await act(async () => {
          result.current.mutate('notif-1')
          expect(result.current.isPending).toBe(true)
        })
      })

      it('should invalidate queries on settle', async () => {
        const invalidateSpy = vi.spyOn(queryClient, 'invalidateQueries')

        const wrapper = ({ children }: { children: React.ReactNode }) => (
          <QueryClientProvider client={queryClient}>{children}</QueryClientProvider>
        )

        const { result } = renderHook(() => useMarkNotificationAsRead(), { wrapper })

        await act(async () => {
          ;(global.fetch as any).mockResolvedValueOnce(
            new Response(null, { status: 200 })
          )
          result.current.mutate('notif-1')
        })

        // Wait for settle
        await new Promise(resolve => setTimeout(resolve, 50))

        expect(invalidateSpy).toHaveBeenCalledWith({ queryKey: ['notifications', 'test-contractor-id'] })
        expect(invalidateSpy).toHaveBeenCalledWith({ queryKey: ['notifications-unread', 'test-contractor-id'] })
      })
    })
    ```

    Key testing points:
    - Mock fetch to return controlled responses
    - Setup QueryClient with test contractor ID
    - Test optimistic update: notification array updated before server response
    - Test unread count decrement
    - Test error handling (isError, mutationError)
    - Test loading state (isPending)
    - Test invalidation on settle (both notifications and unread count queries)
