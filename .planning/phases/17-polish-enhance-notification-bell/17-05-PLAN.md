    This is a comprehensive component test suite covering all visible behavior and user interactions.

    Structure (React Testing Library):

    ```typescript
    import { describe, it, expect, beforeEach, vi } from 'vitest'
    import { render, screen, fireEvent, waitFor } from '@testing-library/react'
    import userEvent from '@testing-library/user-event'
    import NotificationBell from '@/components/notification/NotificationBell'
    import { QueryClient, QueryClientProvider } from '@tanstack/react-query'
    import { ContractorProvider } from '@/contexts/ContractorContext'

    // Mock hooks
    vi.mock('@/lib/query/client', () => ({
      useNotifications: vi.fn(() => ({
        data: [],
        isLoading: false,
        error: null
      })),
      useUnreadCount: vi.fn(() => ({
        data: 0,
        isLoading: false,
        error: null
      })),
      useMarkAllRead: vi.fn(() => ({
        mutate: vi.fn(),
        isPending: false,
        isError: false
      }))
    }))

    // Mock Supabase Realtime
    vi.mock('@supabase/ssr', () => ({
      createBrowserClient: () => ({
        channel: vi.fn(() => ({
          on: vi.fn().mockReturnThis(),
          subscribe: vi.fn().mockReturnThis(),
          unsubscribe: vi.fn().mockReturnThis()
        }))
      })
    }))

    describe('NotificationBell', () => {
      let queryClient: QueryClient

      beforeEach(() => {
        queryClient = new QueryClient({
          defaultOptions: {
            queries: { retry: false },
            mutations: { retry: false }
          }
        })
        vi.clearAllMocks()
      })

      const renderNotificationBell = () => {
        return render(
          <QueryClientProvider client={queryClient}>
            <ContractorProvider>
              <NotificationBell />
            </ContractorProvider>
          </QueryClientProvider>
        )
      }

      it('should render Bell icon', () => {
        renderNotificationBell()
        const bellIcon = screen.getByRole('button', { name: /notification/i })
        expect(bellIcon).toBeInTheDocument()
      })

      it('should display unread count badge when count > 0', () => {
        const { useUnreadCount } = require('@/lib/query/client')
        useUnreadCount.mockReturnValue({ data: 5, isLoading: false, error: null })

        renderNotificationBell()
        expect(screen.getByText('5')).toBeInTheDocument()
      })

      it('should not display badge when unread count is 0', () => {
        const { useUnreadCount } = require('@/lib/query/client')
        useUnreadCount.mockReturnValue({ data: 0, isLoading: false, error: null })

        renderNotificationBell()
        expect(screen.queryByText('0')).not.toBeInTheDocument()
      })

      it('should open popover on button click', async () => {
        renderNotificationBell()
        const bellButton = screen.getByRole('button', { name: /notification/i })

        fireEvent.click(bellButton)

        await waitFor(() => {
          expect(screen.getByRole('dialog', { hidden: false })).toBeInTheDocument()
        })
      })

      it('should close popover when Escape key is pressed', async () => {
        renderNotificationBell()
        const bellButton = screen.getByRole('button', { name: /notification/i })

        fireEvent.click(bellButton)
        await waitFor(() => {
          expect(screen.getByRole('dialog', { hidden: false })).toBeInTheDocument()
        })

        fireEvent.keyDown(document, { key: 'Escape' })
        await waitFor(() => {
          expect(screen.queryByRole('dialog', { hidden: false })).not.toBeInTheDocument()
        })
      })

      it('should display notification list in popover', async () => {
        const { useNotifications } = require('@/lib/query/client')
        useNotifications.mockReturnValue({
          data: [
            { id: '1', title: 'Lead', body: 'New lead', is_read: false, created_at: new Date().toISOString() },
            { id: '2', title: 'Call', body: 'Missed call', is_read: false, created_at: new Date().toISOString() }
          ],
          isLoading: false,
          error: null
        })

        renderNotificationBell()
        const bellButton = screen.getByRole('button', { name: /notification/i })

        fireEvent.click(bellButton)

        await waitFor(() => {
          expect(screen.getByText('Lead')).toBeInTheDocument()
          expect(screen.getByText('Call')).toBeInTheDocument()
        })
      })

      it('should display empty state when no notifications', async () => {
        const { useNotifications } = require('@/lib/query/client')
        useNotifications.mockReturnValue({
          data: [],
          isLoading: false,
          error: null
        })

        renderNotificationBell()
        const bellButton = screen.getByRole('button', { name: /notification/i })

        fireEvent.click(bellButton)

        await waitFor(() => {
          expect(screen.getByText(/no notifications/i)).toBeInTheDocument()
        })
      })

      it('should call useMarkAllRead.mutate() when mark-all-read button is clicked', async () => {
        const { useMarkAllRead, useNotifications } = require('@/lib/query/client')
        const mockMutate = vi.fn()

        useNotifications.mockReturnValue({
          data: [{ id: '1', title: 'Lead', body: 'New lead', is_read: false, created_at: new Date().toISOString() }],
          isLoading: false,
          error: null
        })

        useMarkAllRead.mockReturnValue({
          mutate: mockMutate,
          isPending: false,
          isError: false
        })

        renderNotificationBell()
        const bellButton = screen.getByRole('button', { name: /notification/i })

        fireEvent.click(bellButton)

        await waitFor(() => {
          const markAllReadButton = screen.getByRole('button', { name: /mark all read/i })
          expect(markAllReadButton).toBeInTheDocument()
        })

        fireEvent.click(screen.getByRole('button', { name: /mark all read/i }))
        expect(mockMutate).toHaveBeenCalled()
      })

      it('should update badge count when Realtime notification arrives', async () => {
        const { useUnreadCount } = require('@/lib/query/client')

        // Initial: 0 unread
        useUnreadCount.mockReturnValue({ data: 0, isLoading: false, error: null })

        const { rerender } = renderNotificationBell()

        // Simulate Realtime update: increment to 1
        useUnreadCount.mockReturnValue({ data: 1, isLoading: false, error: null })

        rerender(
          <QueryClientProvider client={queryClient}>
            <ContractorProvider>
              <NotificationBell />
            </ContractorProvider>
          </QueryClientProvider>
        )

        await waitFor(() => {
          expect(screen.getByText('1')).toBeInTheDocument()
        })
      })

      it('should return focus to Bell button when popover closes', async () => {
        renderNotificationBell()
        const bellButton = screen.getByRole('button', { name: /notification/i })

        bellButton.focus()
        expect(document.activeElement).toBe(bellButton)

        fireEvent.click(bellButton)

        await waitFor(() => {
          expect(screen.getByRole('dialog', { hidden: false })).toBeInTheDocument()
        })

        fireEvent.keyDown(document, { key: 'Escape' })

        await waitFor(() => {
          expect(document.activeElement).toBe(bellButton)
        })
      })
    })
    ```

    - Test mark-all-read button (BELL-03)
